#include<stdio.h>

int main() {
    int arr[] = {10, 20, 30, 40, 50}, *p = arr, i;

    printf("array:\n");
    for (i = 0; i < 5; i++)
        printf("%d ", arr[i]);
    
    printf("\npointer:\n");
    for (i = 0; i < 5; i++)
        printf("%d ", *(p + i));
    
    printf("\naddress difference:\n");
    for (i = 0; i < 4; i++)
        printf("%ld bytes\n", (char*)(p + i + 1) - (char*)(p + i));

    return 0;
}
